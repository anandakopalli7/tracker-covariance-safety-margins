def fin(x):
    return x is not None and np.isfinite(x)

def ci95(b):
    b = np.asarray(b, float); f = np.isfinite(b)
    return [float(v) for v in np.percentile(b[f], [2.5, 97.5])] if f.any() else [None, None]


def put_ci(key, b):

def rr_b(D, den):

def ratio_b(a, b):
    a = np.asarray(a, float); b = np.asarray(b, float)
    with np.errstate(divide='ignore', invalid='ignore'):
        return np.where(np.isfinite(b) & (b > 0) & np.isfinite(a), a / b, np.nan)


def cmp(name, mine, ref, tol=0.):
    """Record a comparison with a value of record (exact unless tol > 0)."""
    def one_d(a, b):                       # None and NaN are 'undefined'; two undefined values agree
        a = np.nan if a is None else float(a); b = np.nan if b is None else float(b)
        if not np.isfinite(a) or not np.isfinite(b):
            return 0. if (np.isnan(a) and np.isnan(b)) or a == b else float('inf')
        return abs(a - b)
    if isinstance(mine, (list, tuple)):
        d = max(one_d(a, b) for a, b in zip(mine, ref)) if len(mine) == len(ref) else float('inf')
    else:
        d = one_d(mine, ref)
    CHK[name] = dict(mine=mine, record=ref, abs_diff=d, ok=bool(d <= tol))
    return d


# ------------------------------------------------------------------------------------------------ gridless events, cached per run
class Events:
    """Per run, per (scoring, source): the A9-1(c) gridless events (a9_common_v01.gridless_sources, same lines), kept so that the
    Gridless object of any subset of runs (B6 terciles) is a9_common_v01.Gridless over that subset's per-run tuples."""

    def __init__(self, runs, masks_key):
        self.runs = runs; self.mk = masks_key; self.cache = {}

    def per(self, sc, a, safe_fn):
        if (sc, a) not in self.cache:
            out = []
            for r in self.runs:
                sx = safe_fn(r)
                if a == 'O':
                    sel = L.select_first_safe(sx, r['idx55'])[0]
                    RS = A9.row_stats(r, sx, r[self.mk])
                    T0 = RS[np.arange(sel.size), sel].sum(axis=0)
                    out.append((T0, np.zeros(0), np.zeros((0, len(r[self.mk]), L.NST))))
                else:
                    out.append(A9.run_events(r, r['crit'][a], sx, r[self.mk], A9.GMAX))
            self.cache[(sc, a)] = out
        return self.cache[(sc, a)]

    def G(self, sc, names, safe_fn, idx=None):
        idx = range(len(self.runs)) if idx is None else idx
        return {a: A9.Gridless([self.per(sc, a, safe_fn)[i] for i in idx]) for a in names}

def add_partial(runs, seed):
    """Partial oracles of record: one uniform per stored decision, runs in the given order, decisions in k order."""
    N = int(sum(len(r['k']) for r in runs))
    U = np.random.default_rng(seed).random(N); pos = 0
    for r in runs:
        n = len(r['k']); u = U[pos:pos + n]; pos += n
        for p, nm in zip(PS_, PO):
            r['crit'][nm] = np.where((u < p)[:, None], r['crit']['E'], r['crit']['F0'])
            r['sel'].pop(nm, None)
    return N

def delay_table(rids, evs):
    d = TS.draw_delays(int(sum(int(e.sum()) for e in evs)))

        tau = np.full(ev.size, np.nan); tau[idx] = 0.75 + d[pos:pos + idx.size]
        out[rid] = tau; pos += idx.size
    return out

def h4_parts(st, w):
    """cd_common_v13.H4Stat.share (same lines), returning the sums: num_rec = sum(F_F0 - F_ps), num_like = sum(F_1 - F_ps),
    den = sum(F_F0 - F_E), and n (decisions)."""
    fF0, fE, tri, one = st.fF0, st.fE, st.tri, st.one
    Ta = np.tensordot(w, fF0[:, :, 0, :], axes=(0, 0)); TE = np.tensordot(w, fE[:, :, 0, :], axes=(0, 0))
    c = L.curves_w({'F0': Ta, 'E': TE}, 0); cF0, wF0 = c['F0']; cE, wE = c['E']
    lev_ = L.levels_of(cF0, cE, cF0[0])
    f0 = L.frontier(cF0, wF0, lev_); fe = L.frontier(cE, wE, lev_)
    Tph = [np.tensordot(w, fF0[:, :, pi, :], axes=(0, 0)) for pi in (2, 3, 4)]
    nl = tri.shape[0]; Wv = np.zeros(nl); Sv = np.zeros(nl); Kv = np.zeros(nl)
    for q in range(3):
        g = tri[:, q]; okg = g >= 0
        Wv[okg] += Tph[q][g[okg], 1]; Sv[okg] += Tph[q][g[okg], 2]; Kv[okg] += Tph[q][g[okg], 3]
    n = float(Ta[0, 0])
    with np.errstate(divide='ignore', invalid='ignore'):
        cp = np.where(tri[:, 0] >= 0, 100. * Sv / Kv, np.nan); wp = np.where(tri[:, 0] >= 0, 100. * Wv / n, np.nan)
        ok1 = one >= 0; g1 = np.clip(one, 0, None)
        c1 = np.where(ok1, 100. * Ta[g1, 2] / Ta[g1, 3], np.nan); w1 = np.where(ok1, 100. * Ta[g1, 1] / n, np.nan)
    fps = L.frontier(np.concatenate([[cF0[0]], cp]), np.concatenate([[wF0[0]], wp]), lev_)
    f1 = L.frontier(np.concatenate([[cF0[0]], c1]), np.concatenate([[wF0[0]], w1]), lev_)
    return float(np.sum(f0 - fps)), float(np.sum(f1 - fps)), float(np.sum(f0 - fe)), n

ARMS_REL = ['N', 'NA', 'A1', 'CK3']


def reliance(rs_, sname, pi=0):
    """a9_6_stop_reliance_v01 part (1), same lines, one population (pi); safe set r['safe_x'][sname]."""
    X = C11.tables(rs_, ('X', sname), ['F0', 'E', 'O'] + ARMS_REL)
    Tsum = {a: X[a].sum(axis=0)[:, pi, :] for a in X}
    cur = L.curves_w({a: Tsum[a] for a in Tsum if a != 'O'}, 0)
    cF0, wF0 = cur['F0']; cE, wE = cur['E']
    lev = L.levels_of(cF0, cE, cF0[0])
    n = float(Tsum['F0'][0, 0])
    ent = dict(n=n)
    for a in ['F0', 'E'] + ARMS_REL + ['O']:
        only_stop = 0.; unsafe3 = 0.; unsafe_stop = 0.
        if a == 'O':
            for r in rs_:
                m = r['masks'][pi]; sx = r['safe_x'][sname]; s1 = r['clear1'] > 0.
                sel = L.select_first_safe(sx, r['idx55'])[0]; ar = np.arange(sel.size)
                only_stop += float((m & ~s1[ar, sel] & sx[ar, sel]).sum()); unsafe3 += float((m & ~s1[ar, sel]).sum())
                unsafe_stop += float((m & ~sx[ar, sel]).sum())
            ent[a] = dict(share_safe_only_with_stop=only_stop / n, share_unsafe_3s=unsafe3 / n, share_unsafe_with_stop=unsafe_stop / n)
            continue
        f, ix = A9.frontier_argmin(cur[a][0], cur[a][1], lev)
        if a in ('F0', 'E'):
            cnt = np.bincount(ix[ix >= 0], minlength=len(C11.GRID)).astype(float); cnt0 = 0.; base = a
        else:
            cnt0 = float((ix == 0).sum()); cnt = np.bincount(ix[ix >= 1] - 1, minlength=len(C11.GRID)).astype(float); base = a
        for r in rs_:
            m = r['masks'][pi]; sx = r['safe_x'][sname]; s1 = r['clear1'] > 0.; ar = np.arange(r['n_dec'])
            sA = C11.sel_of(r, base); sF = C11.sel_of(r, 'F0')
            os_ = (~s1[ar[None, :], sA] & sx[ar[None, :], sA]) & m[None, :]
            u3 = (~s1[ar[None, :], sA]) & m[None, :]
            us = (~sx[ar[None, :], sA]) & m[None, :]
            only_stop += float(cnt @ os_.sum(axis=1)); unsafe3 += float(cnt @ u3.sum(axis=1)); unsafe_stop += float(cnt @ us.sum(axis=1))
            if cnt0:
                os0 = (~s1[ar, sF[0]] & sx[ar, sF[0]]) & m; u30 = (~s1[ar, sF[0]]) & m; us0 = (~sx[ar, sF[0]]) & m
                only_stop += cnt0 * float(os0.sum()); unsafe3 += cnt0 * float(u30.sum()); unsafe_stop += cnt0 * float(us0.sum())
        ent[a] = dict(share_safe_only_with_stop=only_stop / L.NLEV / n, share_unsafe_3s=unsafe3 / L.NLEV / n,
                      share_unsafe_with_stop=unsafe_stop / L.NLEV / n)
    return ent
